1. Introducción
Para cumplir con los requerimientos de GrigoriPerelmánApp, se ha desarrollado un backend robusto utilizando Spring Boot, un framework Java que simplifica la creación de aplicaciones empresariales. Este backend expone una API REST que puede ser consumida por una aplicación móvil (Android, iOS o híbrida) y también genera vistas web con Thymeleaf para navegadores. De esta forma, se cubren los tres tipos de módulos solicitados: web, móvil y standalone (mediante un cliente de consola que consume la API). Todo el código sigue estándares de codificación (PascalCase, camelCase, comentarios Javadoc) y está versionado en GitHub.

2. Objetivo
Codificar los módulos del software GrigoriPerelmánApp utilizando Spring Boot como framework, integrando persistencia con JPA/Hibernate, servicios REST, interfaz web con Thymeleaf y soporte para cliente móvil, cumpliendo con los estándares de codificación y buenas prácticas.

3. Framework Seleccionado: Spring Boot
Justificación:

Permite crear rápidamente aplicaciones listas para producción.

Integración nativa con JPA, seguridad, y REST.

Soporte para Thymeleaf (vistas web) y para APIs JSON (móvil).

Amplia comunidad y documentación.

Versión: Spring Boot 3.2.x (requiere Java 17+).

7. Capa de Servicio
UsuarioService (lógica de registro, login, etc.):

java
package com.grigori.app.service;

import com.grigori.app.model.Usuario;
import com.grigori.app.repository.UsuarioRepository;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;
import java.util.Optional;

@Service
public class UsuarioService {
    
    @Autowired
    private UsuarioRepository usuarioRepository;
    
    /**
     * Registra un nuevo usuario si el email no existe.
     * @param nombre nombre completo
     * @param email correo electrónico
     * @param password contraseña en texto plano (se guarda sin hash por simplicidad)
     * @return true si se registró, false si ya existe
     */
    public boolean registrarUsuario(String nombre, String email, String password) {
        if (usuarioRepository.findByEmail(email).isPresent()) {
            return false;
        }
        Usuario usuario = new Usuario(nombre, email, password); // En producción usar BCrypt
        usuarioRepository.save(usuario);
        return true;
    }
    
    /**
     * Valida credenciales de inicio de sesión.
     * @param email correo
     * @param password contraseña
     * @return Optional con el usuario si existe y coincide
     */
    public Optional<Usuario> autenticar(String email, String password) {
        Optional<Usuario> usuarioOpt = usuarioRepository.findByEmail(email);
        if (usuarioOpt.isPresent() && usuarioOpt.get().getPasswordHash().equals(password)) {
            return usuarioOpt;
        }
        return Optional.empty();
    }
}

Controlador de Autenticación (vistas web)
java
package com.grigori.app.controller;

import com.grigori.app.service.UsuarioService;
import jakarta.servlet.http.HttpSession;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Controller;
import org.springframework.ui.Model;
import org.springframework.web.bind.annotation.*;

@Controller
public class AuthController {
    
    @Autowired
    private UsuarioService usuarioService;
    
    // Mostrar formulario de login (GET)
    @GetMapping("/login")
    public String mostrarLogin() {
        return "login";
    }
    
    // Procesar login (POST)
    @PostMapping("/login")
    public String procesarLogin(@RequestParam String email,
                                @RequestParam String password,
                                HttpSession session,
                                Model model) {
        var usuarioOpt = usuarioService.autenticar(email, password);
        if (usuarioOpt.isPresent()) {
            session.setAttribute("usuarioId", usuarioOpt.get().getId());
            session.setAttribute("usuarioNombre", usuarioOpt.get().getNombre());
            return "redirect:/dashboard";
        } else {
            model.addAttribute("error", "Credenciales inválidas");
            return "login";
        }
    }
    
    // Registro (GET y POST análogos)
    @GetMapping("/registro")
    public String mostrarRegistro() {
        return "registro";
    }
    
    @PostMapping("/registro")
    public String procesarRegistro(@RequestParam String nombre,
                                   @RequestParam String email,
                                   @RequestParam String password,
                                   Model model) {
        boolean ok = usuarioService.registrarUsuario(nombre, email, password);
        if (ok) {
            return "redirect:/login?registro=exitoso";
        } else {
            model.addAttribute("error", "El email ya está registrado");
            return "registro";
        }
    }
    
    @GetMapping("/logout")
    public String logout(HttpSession session) {
        session.invalidate();
        return "redirect:/login";
    }
}
Controlador REST para móvil (API)
java
package com.grigori.app.controller;

import com.grigori.app.dto.LoginRequest;
import com.grigori.app.dto.ProblemaRequest;
import com.grigori.app.model.Problema;
import com.grigori.app.model.Usuario;
import com.grigori.app.service.ProblemaService;
import com.grigori.app.service.UsuarioService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Optional;

@RestController
@RequestMapping("/api")
public class ApiController {
    
    @Autowired
    private UsuarioService usuarioService;
    
    @Autowired
    private ProblemaService problemaService;
    
    // Endpoint de login (POST)
    @PostMapping("/login")
    public ResponseEntity<?> login(@RequestBody LoginRequest request) {
        Optional<Usuario> usuario = usuarioService.autenticar(request.getEmail(), request.getPassword());
        if (usuario.isPresent()) {
            Map<String, Object> response = new HashMap<>();
            response.put("id", usuario.get().getId());
            response.put("nombre", usuario.get().getNombre());
            response.put("email", usuario.get().getEmail());
            return ResponseEntity.ok(response);
        } else {
            return ResponseEntity.status(HttpStatus.UNAUTHORIZED).body("Credenciales inválidas");
        }
    }
    
    // Endpoint para registrar usuario (POST)
    @PostMapping("/registro")
    public ResponseEntity<?> registro(@RequestBody Usuario usuario) {
        boolean ok = usuarioService.registrarUsuario(usuario.getNombre(), usuario.getEmail(), usuario.getPasswordHash());
        if (ok) {
            return ResponseEntity.status(HttpStatus.CREATED).body("Usuario registrado");
        } else {
            return ResponseEntity.badRequest().body("Email ya existe");
        }
    }
    
    // Obtener problemas de un usuario (GET)
    @GetMapping("/usuarios/{usuarioId}/problemas")
    public List<Problema> listarProblemas(@PathVariable int usuarioId) {
        return problemaService.listarPorUsuario(usuarioId);
    }
    
    // Guardar nuevo problema (POST)
    @PostMapping("/problemas")
    public ResponseEntity<?> guardarProblema(@RequestBody ProblemaRequest request) {
        Problema problema = new Problema();
        problema.setUsuarioId(request.getUsuarioId());
        problema.setEcuacion(request.getEcuacion());
        problema.setSolucion(request.getSolucion());
        problemaService.guardar(problema);
        return ResponseEntity.status(HttpStatus.CREATED).body(problema);
    }
    
    // Eliminar problema (DELETE)
    @DeleteMapping("/problemas/{id}")
    public ResponseEntity<?> eliminarProblema(@PathVariable int id) {
        problemaService.eliminar(id);
        return ResponseEntity.noContent().build();
    }
}

Se incluye un ejemplo de cliente en Java Standalone (consola) que consume la API mediante HttpURLConnection. Este cliente actúa como módulo standalone.

Cliente Standalone (StandaloneClient.java):

java
package com.grigori.client;

import java.io.*;
import java.net.HttpURLConnection;
import java.net.URL;
import java.util.Scanner;

/**
 * Cliente de consola que consume la API REST de GrigoriPerelmánApp.
 * Demuestra el módulo standalone.
 */
public class StandaloneClient {
    private static final String BASE_URL = "http://localhost:8080/api";
    private static String token = null;
    
    public static void main(String[] args) throws IOException {
        Scanner scanner = new Scanner(System.in);
        System.out.println("=== Cliente GrigoriPerelmánApp (Standalone) ===");
        System.out.print("Email: ");
        String email = scanner.nextLine();
        System.out.print("Contraseña: ");
        String password = scanner.nextLine();
        
        // Llamada a login
        String jsonBody = String.format("{\"email\":\"%s\",\"password\":\"%s\"}", email, password);
        String response = sendPost(BASE_URL + "/login", jsonBody);
        if (response.contains("id")) {
            System.out.println("Login exitoso. Respuesta: " + response);
            // Aquí se podrían listar problemas, etc.
        } else {
            System.out.println("Error de login: " + response);
        }
        scanner.close();
    }
    
    private static String sendPost(String urlStr, String json) throws IOException {
        URL url = new URL(urlStr);
        HttpURLConnection conn = (HttpURLConnection) url.openConnection();
        conn.setRequestMethod("POST");
        conn.setRequestProperty("Content-Type", "application/json");
        conn.setDoOutput(true);
        try(OutputStream os = conn.getOutputStream()) {
            os.write(json.getBytes());
            os.flush();
        }
        int code = conn.getResponseCode();
        BufferedReader br = new BufferedReader(new InputStreamReader(
            (code >= 200 && code < 300) ? conn.getInputStream() : conn.getErrorStream()
        ));
        StringBuilder sb = new StringBuilder();
        String line;
        while ((line = br.readLine()) != null) sb.append(line);
        return sb.toString();
    }
}
